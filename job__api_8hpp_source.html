341 template<typename F, typename S>
342 void ParallelFor(const char* const name, Counter* const counter, const std::size_t start, const std::size_t count, S&& splitter, F&& fn, const QueueMode queue = QueueMode::Default)
343 {
344 job::Dispatch(name, counter, [=, splitter = std::forward<S>(splitter), fn = std::forward<F>(fn)](const job::Ctx& ctx) -> void {
345 if (count > 1u && splitter(count))
346 {
347 const std::size_t left_count = count / 2;
348 const std::size_t right_count = count - left_count;
349
350 job::ParallelFor(ctx.task_name, ctx.task_counter, start + 0, left_count, splitter, fn, queue);
351 job::ParallelFor(ctx.task_name, ctx.task_counter, start + left_count, right_count, splitter, fn, queue);
352 }
353 else
354 {
355 for (std::size_t offset = 0u; offset < count; ++offset)
356 {
357 fn(ctx, start + offset);
358 }
359 } }, queue);
360 }

391 template<typename Splitter, typename Reducer>
392 void ParallelReduce(const char* const name, Counter* const counter, const std::size_t start, const std::size_t count, Splitter&& splitter, Reducer&& reduce, const QueueMode queue = QueueMode::Default)
393 {
394 const auto ParallelReduce_Impl = [=, splitter = std::forward<Splitter>(splitter), reduce = std::forward<Reducer>(reduce)](const job::Ctx& ctx) -> void {
395 // NOTE(SR):
396 // Could also have a stride that increases each step.
397 // This would be bad for Cuda GPU (Shared Memory Bank Conflict)
398 // But good on CPU with better locality.
399 // https://developer.download.nvidia.com/assets/cuda/files/reduction.pdf
400
401 std::size_t count_left = count;
402
403 while (count_left > 1)
404 {
405 const std::size_t stride = count_left / 2;
406
407 const auto ReduceRange = [stride, &reduce](const job::Ctx& ctx, const std::size_t index) -> void {
408 reduce(ctx, index, index + stride);
409 };
410
411 Counter c{};
412 ParallelFor(name, &c, start, stride, splitter, ReduceRange, queue);
413 WaitOn(c);
414
415 if ((count_left & 1) != 0)
416 {
417 reduce(ctx, start, start + count_left - 1);
418 }
419
420 count_left = stride;
421 } };
422
423 job::Dispatch(name, counter, ParallelReduce_Impl, queue);
424 }
